cmake_minimum_required(VERSION 3.16)
project(maelys_bundle_example C)
set(CMAKE_C_STANDARD 11)
set(MAELYS_SDK_PREFIX "" CACHE PATH "Installed Maelys SDK prefix (required)")
option(MAELYS_SDK_SHARED "Test against the shared engine" OFF)
if(NOT MAELYS_SDK_PREFIX)
    message(FATAL_ERROR "Set MAELYS_SDK_PREFIX to a fresh SDK installation")
endif()
find_path(SDK_INCLUDE maelys/datalog_extension.h PATHS "${MAELYS_SDK_PREFIX}/include" NO_DEFAULT_PATH)
find_path(KIT_INCLUDE maelys_conformance.h PATHS "${MAELYS_SDK_PREFIX}/share/maelys-datalog/conformance" NO_DEFAULT_PATH)
if(MAELYS_SDK_SHARED)
    set(engine_name maelys_datalog_shared)
else()
    set(engine_name maelys_datalog)
endif()
find_library(SDK_LIBRARY NAMES ${engine_name} PATHS "${MAELYS_SDK_PREFIX}/lib" "${MAELYS_SDK_PREFIX}/lib64" NO_DEFAULT_PATH)
if(NOT SDK_INCLUDE OR NOT KIT_INCLUDE OR NOT SDK_LIBRARY)
    message(FATAL_ERROR "Incomplete installed SDK or conformance kit")
endif()
add_library(extension STATIC src/extension.c)
target_include_directories(extension PUBLIC include "${SDK_INCLUDE}")
target_compile_options(extension PRIVATE -Wall -Wextra -Werror)
add_executable(conformance tests/conformance.c)
target_include_directories(conformance PRIVATE "${KIT_INCLUDE}")
target_compile_options(conformance PRIVATE -Wall -Wextra -Werror)
target_link_libraries(conformance PRIVATE extension "${SDK_LIBRARY}")
enable_testing()
add_test(NAME bundle_conformance COMMAND conformance)
